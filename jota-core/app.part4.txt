function product3DCard(p){const calc=price3D(p);return `<div class="card"><span class="pill good">${esc(p.stage)}</span><h3 style="margin-top:8px">${esc(p.name)}</h3><p>${esc(p.problem)}</p><div class="grid g2"><div class="notice"><b>Custo</b><br>${euro(calc.cost)}</div><div class="notice green"><b>Preço sugerido</b><br>${euro(calc.suggested)}</div></div><p class="mini">Canais: ${calc.channels.join(' · ')}</p><button class="btn small" onclick="app.edit3D('${p.id}')">Editar</button></div>`}
function price3D(p){const material=Number(p.material||0),hours=Number(p.hours||0),human=Number(p.human||0),pack=Number(p.pack||0),fail=Number(p.failPct||10)/100;const machine=hours*Number(p.machineRate||1.2),labor=human*Number(p.laborRate||12);const cost=(material+machine+labor+pack)*(1+fail);let suggested=Math.max(cost*2.2,cost+8);if(p.digital)suggested=Math.max(Number(p.digitalPrice||6.9),6.9);const channels=p.digital?['Cults','Etsy digital','site próprio mais tarde']:['local/POD','Etsy se design próprio','B2B direto'];return {cost,suggested,channels}}
